    """
    根据标题生成id: 空格→"-" + 小写

    """

        while f"{base_id}-{suffix}" in existing_ids:
            suffix += 1
        
        unique_id = f"{base_id}-{suffix}"

        return unique_id, warning
    else:
        return base_id, None

```python
import re
import unicodedata

def generate_id_from_title(title, existing_ids):
    """
    根据标题生成id: 
    1. 法语字符→英语字母 (é→e, è→e, à→a, ç→c)
    2. 空格→"-"
    3. 转换为小写
    4. 移除特殊字符
    5. 检查重复，自动添加后缀
    """
    # 1. 法语字符转英语字母 (Unicode规范化)
    normalized = unicodedata.normalize('NFKD', title.strip())
    ascii_text = normalized.encode('ASCII', 'ignore').decode('ASCII')
    
    # 如果没有ASCII字符（全是非ASCII字符），使用原始文本
    if not ascii_text:
        ascii_text = title.strip()
    
    # 2. 空格→"-"，转换为小写
    base_id = ascii_text.lower().replace(' ', '-')
    
    # 3. 移除特殊字符，只保留字母数字和连字符
    base_id = re.sub(r'[^a-z0-9-]', '', base_id)
    
    # 4. 确保不以连字符开头或结尾
    base_id = base_id.strip('-')
    
    # 5. 检查重复
    if base_id in existing_ids:
        # 自动生成唯一id
        suffix = 2
        while f"{base_id}-{suffix}" in existing_ids:
            suffix += 1
        
        unique_id = f"{base_id}-{suffix}"
        warning = f"警告: 标题'{title}'生成的id已存在，已自动调整为'{unique_id}'"
        return unique_id, warning
    else:
        return base_id, None
```

2. **关键需求**: 
   - 检测哪些页面引用了被移动的页面
   - 提供引用更新或重定向机制
   - 通知相关人员和管理员
3. **复杂性**: WebBot页面之间存在多种引用关系（父子关系、语言配对、内容链接等）

#### **技术影响**:
- 需要扩展数据库设计，增加引用管理表
- 需要实现引用检测和更新算法
- 需要设计用户友好的通知和操作界面

### 🛠️ **解决方案设计**

#### **1. 引用关系表设计**:
```sql
CREATE TABLE webbot_page_references (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    source_page_id TEXT NOT NULL REFERENCES webbot_page(id),
    target_page_id TEXT NOT NULL REFERENCES webbot_page(id),
    reference_type TEXT NOT NULL,  -- 'link', 'parent', 'lang_pair', 'media', 'custom'
    context_path TEXT,            -- 在内容中的位置路径
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(source_page_id, target_page_id, reference_type)
);

CREATE TABLE webbot_url_redirects (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    old_url_path TEXT NOT NULL UNIQUE,  -- 如 "/en/old-page-id"
    new_url_path TEXT NOT NULL,         -- 如 "/en/new-page-id"
    redirect_type TEXT DEFAULT 'permanent',  -- 301或302
    created_by INTEGER REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP
);
```

- **内容解析**: 从HTML内容中提取内部链接 (`/<lang>/<page-id>`)

- **自动更新**: 页面内容变更时自动更新引用关系

```
页面重命名请求 →

2. 影响分析报告 → 显示需要更新的页面数量
3. 提供操作选项:
   a) 自动更新所有引用链接
   b) 创建URL重定向
   c) 仅生成待办事项列表
4. 执行选择的操作
5. 记录变更历史和发送通知
```

#### **4. API扩展设计**:
- `GET /pages/{page_id}/references` - 获取页面引用关系
- `POST /pages/{page_id}/check-rename-impact` - 检查重命名影响

## 🎯 **HTML引用澄清与方案简化** (2026-03-22 12:50 EDT / 消息ID 812)

### 📋 **用户关键澄清**
**用户消息**: "HTML内容中 never reference to parent_id, other_lang_page_id"

#### **核心理解**:
1. **重要澄清**: HTML内容中永远不会直接引用 `parent_id` 或 `other_lang_page_id` 这些数据库字段
2. **引用机制**: 页面间的引用通过生成的URL (`/<language_code>/<page_id>`) 来实现
3. **设计影响**: 引用管理方案可以大幅简化

### 🔧 **引用管理方案简化**

#### **引用检测范围调整**:
**需要追踪的引用**:
1. ✅ **HTML内容中的URL链接**: `<a href="/en/about-us">` (显式内容链接)
2. ✅ **生成的导航结构**: 基于`parent_id`自动生成的菜单链接
3. ✅ **语言切换链接**: 基于`other_lang_page_id`自动生成的语言切换链接

**不需要追踪的引用**:
1. ❌ **数据库字段的直接引用**: `parent_id`, `other_lang_page_id` 字段本身
2. ❌ **元数据字段关联**: 除非显式转换为URL在内容中使用
3. ❌ **潜在的关联关系**: 仅追踪实际在内容或界面中使用的引用

#### **简化后的引用表设计**:
```sql
CREATE TABLE webbot_page_url_references (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    source_page_id TEXT NOT NULL REFERENCES webbot_page(id),
    target_url_path TEXT NOT NULL,  -- 如 "/en/about-us"
    target_page_id TEXT,            -- 可空，如果URL能解析到有效页面
    reference_context TEXT,         -- 'content_link', 'nav_generated', 'lang_switch'
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    INDEX idx_refs_target_url (target_url_path),
    INDEX idx_refs_source (source_page_id)
);
```

#### **引用检测流程简化**:
```
页面保存/更新时：
1. 解析HTML内容 → 提取所有内部URL链接
2. 基于parent_id生成导航URL → 记录为系统生成的引用
3. 基于other_lang_page_id生成语言切换URL → 记录为系统生成的引用
4. 所有引用保存到 webbot_page_url_references 表
```

#### **页面重命名影响检测**:
```
页面ID改变时：
1. 生成新旧URL映射：/en/old-id → /en/new-id
2. 查询引用表：SELECT * FROM webbot_page_url_references WHERE target_url_path = '/en/old-id'
3. 返回所有引用该URL的页面列表
4. 提供更新选项：自动更新内容链接，重新生成导航和语言切换链接
```

### 🎯 **方案优势**

#### **1. 更简单**:
- 无需解析数据库字段的潜在引用
- 只处理实际存在的URL链接
- 引用关系基于实际使用，而非潜在关联

#### **2. 更准确**:
- 检测的是实际内容中的引用
- 避免误报（如未使用的parent_id关系）
- 反映真实的页面依赖关系

#### **3. 更高效**:
- 引用表更小，查询更快
- 更新逻辑更直接
- 维护成本更低

#### **4. 更符合用户工作流**:
