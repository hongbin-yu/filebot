  ```typescript
    skipIntermediate: boolean;        // true: 不保留中间值
  }
  ```

```typescript
interface Command {
  id: string;
  type: string;
  execute(): void;
  undo(): void;
  redo(): void;
  merge?(nextCommand: Command): boolean; // 命令合并
}
```

#### **4. 历史管理器核心逻辑**：
```typescript
class HistoryManager {
  private undoStack: Command[] = [];
  private redoStack: Command[] = [];
  private currentTransaction: Transaction | null = null;

  beginTransaction(transaction: Omit<Transaction, 'id' | 'startTime'>) { ... }
  
  // 结束事务（关闭对话框时）
  commitTransaction(finalState: ComponentSnapshot) { ... }
  
  // 执行命令（自动处理合并）
  execute(command: Command) { ... }
  
  undo() { ... }
  redo() { ... }
}
```
